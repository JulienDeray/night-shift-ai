    formatSuccessNotification:
    - input: task + result { status: "SUCCESS", totalDurationMs: 222000, finalOutput: "Created MR !42\nWith details" } -> title: "code-agent done: nightly-refactor", body contains "code-agent", "3m 42s", "Created MR !42" (first line only), priority: 3, tags contain a success emoji
    - input: result with finalOutput as object { summary: "Refactored 3 files" } -> body contains "Refactored 3 files"

    - input: result with totalDurationMs: 3700000 -> duration formatted as "1h 1m"
    - input: result with totalDurationMs: 45000 -> duration formatted as "45s"

    formatFailureNotification:
    - input: task + result { status: "FATAL", error: "TypeError: x is not a function\n    at foo.ts:12\n    at bar.ts:34", failedStepIndex: 1, perStep: [{name:"init",...},{name:"execute",...}] } -> title: "code-agent FAILED: nightly-refactor", body contains "Step 'execute' failed", "TypeError: x is not a function" (no stack frames), priority: 4, tags contain a failure emoji
    - input: result with failedStepIndex: undefined -> body contains "unknown step"
    - input: result with failedStepIndex: 5, perStep length 2 -> body contains "step 5" (out of bounds fallback)
    - input: result with error: undefined -> body contains "Unknown error"
