  manifest: z.infer<typeof ManifestSchema>,
  bead: ManifestBead,
  agentEnv: ResolvedEnvVar[],

3. Modify the bead execution loop in `run()` to support retry. The key design:

**Retry semantics:** A bead with `retry` config declares "if MY output indicates failure (via a `passed: false` pattern in the output), retry from `retryFrom` bead up to `maxAttempts` total times." The engine detects `passed === false` in the bead's validated output to trigger retry. This is a generic convention: any bead whose output schema has a `passed` boolean field can be a retry trigger.

**Implementation approach — replace the simple for loop with retry-aware logic:**

```typescript
// Track retry state
let retryCount = 0;
let retryFromIndex = -1;

// Track beadOutputs for the result
const beadOutputs: Record<string, unknown> = {};

let i = 0;
while (i < manifest.beads.length) {
  const bead = manifest.beads[i];
  ctx = { ...ctx, currentBead: bead };
  const beadStart = Date.now();

  // ... existing logging ...

  try {
    const factory = this.registry.resolve(bead.type);
    const plugin = factory(bead, manifest);
    const output = await plugin.execute(ctx);
    const rawOutput = output.rawOutput;
    const parsed = validateBeadOutput(rawOutput, bead.compiledOutputSchema, bead.name);
    const durationMs = Date.now() - beadStart;

    // Store bead output
    beadOutputs[bead.name] = parsed;

    // Accumulate context (existing logic)
    ctx = { ...ctx, previousBeads: { ...ctx.previousBeads, [bead.name]: { output: parsed, rawOutput } } };
    ctx = { ...ctx, variables: buildTemplateVars(builtIns, manifest.variables, configOverrides ?? {}, ctx.previousBeads) };

    perBead.push({ name: bead.name, status: "SUCCESS", durationMs });

    // CHECK RETRY TRIGGER: if bead has retry config and output has passed === false
    if (bead.retry && typeof parsed === "object" && parsed !== null && "passed" in parsed && (parsed as Record<string, unknown>).passed === false) {
      retryCount++;
      if (retryCount <= bead.retry.maxAttempts) {
        // Find retryFrom bead index
        retryFromIndex = manifest.beads.findIndex(b => b.name === bead.retry!.retryFrom);

        this.logger.info("Bead triggered retry", {
          runId,
          bead: bead.name,
          retryFrom: bead.retry.retryFrom,
          attempt: retryCount,
          maxAttempts: bead.retry.maxAttempts,
        });

        // Inject retry_error into variables for the retryFrom bead
        const errorDetails = (parsed as Record<string, unknown>).error_details ?? "";
        ctx = {
          ...ctx,
          variables: {
            ...ctx.variables,
            retry_error: String(errorDetails),
          },
        };

        // Reset working directory (git reset --hard HEAD)
        await this.resetWorkDir(ctx.workDir);

        // Jump back to retryFrom bead
        i = retryFromIndex;
        continue;
      }
      // Max retries exhausted — fall through to normal progression
      this.logger.warn("Retry exhausted", {
        runId,
        bead: bead.name,
        retryCount,
        maxAttempts: bead.retry.maxAttempts,
      });
    }

    // Clear retry state on successful progression past the retry trigger bead
    if (!bead.retry) {
      retryCount = 0;
    }

    // ... existing success logging ...

    i++;
  } catch (err) {
    // ... existing error handling (mark SKIPPED, cleanup, return) ...
    // Add beadOutputs to the returned result
  }
}
```
